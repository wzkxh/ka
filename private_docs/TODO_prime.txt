int64_t is_prime(int64_t n)
{
  if( n<4 ) return n>1;
  if( (n & 1) == 0 || n % 3 == 0 ) return 0;
  for( int64_t i = 5; i*i <= n; i += 6 )
  {
    if( n%i == 0 || n%(i+2) == 0 ) return 0;
  }
  return 1;
}
